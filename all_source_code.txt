import React, { useState, useEffect } from 'react';
import { Sliders, ArrowRight } from 'lucide-react';
import { Card, CardHeader, CardTitle, CardContent } from '../ui/Card';
import { calculateRiskScore } from '../../utils/utils';
import { RiskBadge } from '../ui/RiskBadge';

export function WhatIfSimulator({ initialFinancials }) {
  const [financials, setFinancials] = useState(initialFinancials);
  const [currentScore, setCurrentScore] = useState(null);
  const [newScore, setNewScore] = useState(null);

  useEffect(() => {
    const baseResult = calculateRiskScore(initialFinancials);
    setCurrentScore(baseResult);
    const newResult = calculateRiskScore(financials);
    setNewScore(newResult);
  }, [initialFinancials, financials]);

  const handleChange = (e) => {
    const { name, value } = e.target;
    setFinancials(prev => ({ ...prev, [name]: Number(value) }));
  };

  if (!currentScore || !newScore) return null;

  const scoreDiff = newScore.score - currentScore.score;

  return (
    <Card className="border-primary-200 dark:border-primary-900 shadow-md">
      <CardHeader className="bg-primary-50/50 dark:bg-primary-900/10 border-b border-primary-100 dark:border-primary-900/50">
        <CardTitle className="flex items-center text-primary-800 dark:text-primary-300">
          <Sliders size={20} className="mr-2" />
          What-If Risk Simulator
        </CardTitle>
      </CardHeader>
      <CardContent className="p-6">
        <div className="grid grid-cols-1 md:grid-cols-2 gap-8">
          <div className="space-y-6">
            <div>
              <div className="flex justify-between mb-1">
                <label className="text-sm font-medium">Credit Score</label>
                <span className="text-sm font-bold">{financials.creditScore}</span>
              </div>
              <input 
                type="range" 
                name="creditScore" 
                min="300" max="850" 
                value={financials.creditScore} 
                onChange={handleChange}
                className="w-full h-2 bg-slate-200 dark:bg-slate-700 rounded-lg appearance-none cursor-pointer accent-primary-600"
              />
            </div>
            <div>
              <div className="flex justify-between mb-1">
                <label className="text-sm font-medium">Monthly Income ($)</label>
                <span className="text-sm font-bold">{financials.income}</span>
              </div>
              <input 
                type="range" 
                name="income" 
                min="1000" max="25000" step="500"
                value={financials.income} 
                onChange={handleChange}
                className="w-full h-2 bg-slate-200 dark:bg-slate-700 rounded-lg appearance-none cursor-pointer accent-primary-600"

            </div>

export function A11yPanel() {
  const [isOpen, setIsOpen] = useState(false);
  const panelRef = useRef(null);
  
  const { 
    highContrast, toggleHighContrast,
    textSize, cycleTextSize,
    reduceMotion, toggleReduceMotion,
    resetA11y
  } = useA11y();

  // Close panel on outside click
  useEffect(() => {
    function handleClickOutside(event) {
      if (panelRef.current && !panelRef.current.contains(event.target)) {
        setIsOpen(false);
      }
    }
    if (isOpen) {
      document.addEventListener('mousedown', handleClickOutside);
    }
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, [isOpen]);

  const getTextSizeLabel = () => {
    if (textSize === 'large') return 'Large Text';
    if (textSize === 'xlarge') return 'Extra Large';
    return 'Normal Text';
  };

  return (
    <div className="relative" ref={panelRef}>
      <button 
        onClick={() => setIsOpen(!isOpen)}
        className="p-2 rounded-full hover:bg-slate-100 dark:hover:bg-slate-800 text-slate-600 dark:text-slate-300 transition-colors focus:ring-2 focus:ring-primary-500 outline-none"

import React, { createContext, useContext, useState, useEffect } from 'react';

const A11yContext = createContext();

export function A11yProvider({ children }) {
  const [highContrast, setHighContrast] = useState(() => {
    return localStorage.getItem('a11y-high-contrast') === 'true';
  });
  
  const [textSize, setTextSize] = useState(() => {
    return localStorage.getItem('a11y-text-size') || 'normal';
  });
  
  const [reduceMotion, setReduceMotion] = useState(() => {
    // Check OS preference first, then local storage
    const osPrefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    const stored = localStorage.getItem('a11y-reduce-motion');
    return stored !== null ? stored === 'true' : osPrefersReduced;
  });

  useEffect(() => {
    localStorage.setItem('a11y-high-contrast', highContrast);
    const root = window.document.documentElement;
    if (highContrast) {
      root.classList.add('high-contrast');
    } else {
      root.classList.remove('high-contrast');
    }
  }, [highContrast]);

  useEffect(() => {
    localStorage.setItem('a11y-text-size', textSize);
    const root = window.document.documentElement;
    root.classList.remove('text-size-normal', 'text-size-large', 'text-size-xlarge');
    root.classList.add(`text-size-${textSize}`);
  }, [textSize]);

  useEffect(() => {
    localStorage.setItem('a11y-reduce-motion', reduceMotion);
    const root = window.document.documentElement;
    if (reduceMotion) {
      root.classList.add('reduce-motion');
    } else {
      root.classList.remove('reduce-motion');
    }
  }, [reduceMotion]);

  const toggleHighContrast = () => setHighContrast(prev => !prev);
  
  const cycleTextSize = () => {
    setTextSize(prev => {
      if (prev === 'normal') return 'large';
      if (prev === 'large') return 'xlarge';
      return 'normal';
    });
  };

  const toggleReduceMotion = () => setReduceMotion(prev => !prev);

  const resetA11y = () => {
    setHighContrast(false);
    setTextSize('normal');
    setReduceMotion(window.matchMedia('(prefers-reduced-motion: reduce)').matches);
  };

  return (
    <A11yContext.Provider value={{
      highContrast, toggleHighContrast,
      textSize, cycleTextSize,
      reduceMotion, toggleReduceMotion,
      resetA11y
    }}>
      {children}
    </A11yContext.Provider>
  );
}

export function useA11y() {
  const context = useContext(A11yContext);
  if (context === undefined) {

import React, { createContext, useContext, useState, useEffect } from 'react';

const ThemeContext = createContext();

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState(() => {
    const savedTheme = localStorage.getItem('theme');
    return savedTheme || 'light';
  });

  useEffect(() => {
    localStorage.setItem('theme', theme);
    const root = window.document.documentElement;
    if (theme === 'dark') {
      root.classList.add('dark');
    } else {
      root.classList.remove('dark');
    }
  }, [theme]);

  const toggleTheme = () => {
    setTheme((prev) => (prev === 'light' ? 'dark' : 'light'));
  };

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme() {
  const context = useContext(ThemeContext);
  if (context === undefined) {
    throw new Error('useTheme must be used within a ThemeProvider');
  }
  return context;
}

// Generate demo risk score based on mock data
export function calculateRiskScore(financials) {
  let score = 100;
  const factors = [];

  // Credit Score Factor
  if (financials.creditScore >= 750) {
    factors.push({ name: 'Credit Score', impact: 'Positive', level: 'Strong', desc: 'Excellent credit history' });
  } else if (financials.creditScore >= 650) {
    score -= 10;
    factors.push({ name: 'Credit Score', impact: 'Neutral', level: 'Moderate', desc: 'Fair credit history' });
  } else {
    score -= 30;
    factors.push({ name: 'Credit Score', impact: 'Negative', level: 'High Risk', desc: 'Poor credit history' });
  }

  // Debt-to-Income (DTI) Ratio = (Debt / Income) * 100
  const dti = financials.income > 0 ? (financials.debt / financials.income) * 100 : 100;
  if (dti <= 30) {
    factors.push({ name: 'Debt-to-Income', impact: 'Positive', level: 'Strong', desc: 'Low debt relative to income' });
  } else if (dti <= 50) {
    score -= 15;
    factors.push({ name: 'Debt-to-Income', impact: 'Neutral', level: 'Moderate', desc: 'Moderate debt burden' });
  } else {
    score -= 25;
    factors.push({ name: 'Debt-to-Income', impact: 'Negative', level: 'High Risk', desc: 'High debt burden' });
  }

  // Credit Utilization
  if (financials.utilization <= 30) {
    factors.push({ name: 'Credit Utilization', impact: 'Positive', level: 'Strong', desc: 'Responsible credit usage' });
  } else if (financials.utilization <= 60) {
    score -= 10;
    factors.push({ name: 'Credit Utilization', impact: 'Neutral', level: 'Moderate', desc: 'Moderate credit usage' });
  } else {
    score -= 20;
    factors.push({ name: 'Credit Utilization', impact: 'Negative', level: 'High Risk', desc: 'High credit usage' });
  }

  // Late Payments
  if (financials.latePayments === 0) {
    factors.push({ name: 'Payment History', impact: 'Positive', level: 'Strong', desc: 'No recent late payments' });
  } else if (financials.latePayments <= 2) {
    score -= 15;
    factors.push({ name: 'Payment History', impact: 'Neutral', level: 'Moderate', desc: 'Some recent late payments' });
  } else {
    score -= 30;
    factors.push({ name: 'Payment History', impact: 'Negative', level: 'High Risk', desc: 'Frequent late payments' });
  }

  // Cap score between 0 and 100
  score = Math.max(0, Math.min(100, Math.round(score)));

  let riskLevel = 'HIGH RISK';
  if (score >= 70) riskLevel = 'LOW RISK';
  else if (score >= 40) riskLevel = 'MEDIUM RISK';

  return { score, riskLevel, factors };
}

import React from 'react';
import { BrowserRouter, Routes, Route, Navigate } from 'react-router-dom';
import { ThemeProvider } from './contexts/ThemeContext';
import { A11yProvider } from './contexts/A11yContext';
import { Layout } from './components/layout/Layout';

// Pages placeholders
import { Landing } from './pages/Landing';
import { Dashboard } from './pages/Dashboard';
import { Assessment } from './pages/Assessment';
import { Result } from './pages/Result';
import { Customers } from './pages/Customers';
import { CustomerProfile } from './pages/CustomerProfile';
import { Insights } from './pages/Insights';
import { About } from './pages/About';

function App() {
  return (
    <ThemeProvider>
      <A11yProvider>
        <BrowserRouter>
          <Routes>
            <Route path="/" element={<Landing />} />
            <Route element={<Layout />}>
              <Route path="/dashboard" element={<Dashboard />} />
              <Route path="/assessment" element={<Assessment />} />
              <Route path="/result" element={<Result />} />
              <Route path="/customers" element={<Customers />} />
              <Route path="/customers/:id" element={<CustomerProfile />} />
              <Route path="/insights" element={<Insights />} />
              <Route path="/about" element={<About />} />
            </Route>
            <Route path="*" element={<Navigate to="/" replace />} />
          </Routes>
        </BrowserRouter>
      </A11yProvider>
    </ThemeProvider>
  );
}

export default App;
